% Shared appendix order, with normal in-document labels.
\makeatletter
\newcounter{gaffappendixorder}
\newcommand{\gaffDeclareAppendix}[2]{%
  \stepcounter{gaffappendixorder}%
  \expandafter\xdef\csname gaff@number@#1\endcsname{\Alph{gaffappendixorder}}%
  \expandafter\gdef\csname gaff@title@#1\endcsname{#2}%
}
\newcommand{\GAFFAppendixSection}[1]{%
  \section{\csname gaff@title@#1\endcsname}%
  \edef\gaff@actual{\Alph{section}}%
  \edef\gaff@expected{\csname gaff@number@#1\endcsname}%
  \ifx\gaff@actual\gaff@expected\else
    \PackageError{gaff-appendices}{Appendix order mismatch for #1}%
      {Keep appendix_order.tex and supplementary section order synchronized.}%
  \fi
  \label{#1}%
}
